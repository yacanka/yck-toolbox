cmake_minimum_required(VERSION 3.20)
project(Awaken VERSION 1.0.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)

function(awaken_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    endif()
endfunction()

if(WIN32)
    enable_language(RC)
    add_executable(awaken WIN32 awaken.c awaken.rc)
    target_compile_definitions(awaken PRIVATE _WIN32_WINNT=0x0601)
    target_link_libraries(awaken PRIVATE comctl32 advapi32 shell32 gdi32 user32)
    awaken_warnings(awaken)
    if(MSVC)
        target_link_options(awaken PRIVATE /MANIFEST:NO /DYNAMICBASE /NXCOMPAT)
    endif()
    if(MINGW)
        target_link_options(awaken PRIVATE -static -Wl,--dynamicbase,--nxcompat)
    endif()
    install(TARGETS awaken RUNTIME DESTINATION .)
    install(FILES README.md DESTINATION .)
endif()

if(BUILD_TESTING)
    add_executable(awaken_logic_tests tests/test_logic.c)
    awaken_warnings(awaken_logic_tests)
    add_test(NAME logic COMMAND awaken_logic_tests)
    if(WIN32)
        add_executable(awaken_windows_tests tests/test_windows.c)
        target_compile_definitions(awaken_windows_tests PRIVATE _WIN32_WINNT=0x0601)
        target_link_libraries(awaken_windows_tests PRIVATE comctl32 advapi32 shell32 gdi32 user32)
        awaken_warnings(awaken_windows_tests)
        add_test(NAME windows_lifecycle COMMAND awaken_windows_tests)
    endif()
endif()
